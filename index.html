// One bad row won't fail all records
public with sharing class PaymentService {
  public static void save(
      List<Payment__c> rows) {
    List<Database.SaveResult> res =
      Database.update(rows, false,
        AccessLevel.USER_MODE);
    for (Integer i = 0; i < res.size(); i++) {
      if (!res[i].isSuccess()) {
        Log.error(rows[i].Id,
          res[i].getErrors()[0].getMessage());
      }
    }
  }
}
